تمرير الكائنات كمعاملات للدوال (Passing Objects as Arguments)

بص يا هندسة، الكلاس في الـ C++ هو في الأصل **User-Defined Data Type** (نوع بيانات من صنع المبرمج)؛ يعني زي ما بنقدر نمرر رقم صحيح int أو حرف char لدالة، نقدر كمان نمرر كائن كامل (**Object**) للدالة كـ Argument! وده مفيد جداً لما تحب تعمل دالة تقارن بين طالبين، أو تجمع مسافتين هندسيتين.

طرق تمرير الكائنات للدوال والفرق الجوهري بينها:

نوع التمرير (Passing Mechanism) طريقة الاستدعاء بالكود تأثيره على الميموري والسرعة إمكانية التعديل على الكائن الأصلي
Pass by Value void display(Distance d) بطيء نسبياً: بيعمل نسخة جديدة كاملة من الكائن في الرامات باستدعاء الـ Copy Constructor. مستحيل: أي تعديل جوة الدالة بيأثر على النسخة فقط، والـ Object الأصلي في الـ main مش بيتغير.
Pass by Reference / Pointer void update(Distance &d) سريع جداً: بيمرر عنوان الكائن مباشرة بدون نسخ أي بيانات في الرامات! ممكن وسهل: أي تعديل على المتغيرات جوة الدالة هيعدل الكائن الأصلي في الـ main فوراً.
Pass by Const Reference void show(const Distance &d) أقصى سرعة + حماية كاملة: بيمرر العنوان بدون نسخ لتوفير الميموري، ومع حماية كاملة. ممنوع تماماً: كلمة const بتخلي الـ compiler يطلع error لو الدالة حاولت تعدل في الكائن.

نصيحة المهندس الكبير ليك

يا هندسة، لما الكلاس بتاعك يكون مليان متغيرات ونصوص وعناصر كتير، **ابعد تماماً عن الـ Pass by Value**! لأن نسخ كائن حجمه كبير بياخد وقت وبيستهلك الرامات على الفاضي. الأصح والأفضل دايماً إنك تستخدم **Pass by Reference**، ولو مش محتاج تعدل عليه حط كلمة **const** عشان تحمي الداتا وتضمن أمان الكود وسرعته القصوى.

التطبيق العملي: كلاس الأبعاد مع دالة دمج المسافات خارجية

pass_objects.cpp
#include <iostream>
using namespace std;

class Distance {
private:
    int feet;
    float inches;

public:
    Distance() : feet(0), inches(0.0) {}
    Distance(int ft, float in) : feet(ft), inches(in) {}

    void setDistance(int ft, float in) {
        feet = ft;
        inches = in;
    }

    void showDistance() const {
        cout << feet << "\' - " << inches << "\"" << endl;
    }

    // Getters
    int getFeet() const { return feet; }
    float getInches() const { return inches; }
};

// دالة خارجية تستقبل كائنين بالـ Const Reference وكائن تالت بالـ Reference لتعديله
// بص يا هندسة، هنا طبقنا المعادلة الصح وعالجنا الكسور الذكية (لو الإنش عدى 12)
void addDistances(const Distance &d1, const Distance &d2, Distance &result) {
    float totalInches = d1.getInches() + d2.getInches();
    int extraFeet = 0;

    // معالجة الكسور الذكية: لو الإنشات عدت أو ساوت 12
    if (totalInches >= 12.0) {
        extraFeet = (int)(totalInches / 12.0);
        totalInches = totalInches - (extraFeet * 12.0);
    }

    int totalFeet = d1.getFeet() + d2.getFeet() + extraFeet;
    
    // تعديل كائن النتيجة الأصلي في الـ main لأننا ممررينه بـ &
    result.setDistance(totalFeet, totalInches);
}

int main() {
    Distance d1(5, 9.5); // 5 قدم و 9.5 إنش
    Distance d2(4, 7.2); // 4 قدم و 7.2 إنش
    Distance sum;        // كائن لحفظ النتيجة

    cout << "Distance 1: ";
    d1.showDistance();
    cout << "Distance 2: ";
    d2.showDistance();

    // استدعاء دالة الدمج وتمرير الكائنات
    addDistances(d1, d2, sum);

    cout << "\nTotal Sum: ";
    sum.showDistance(); // المفروض تطلع 10 قدم و 4.7 إنش لأن (9.5 + 7.2 = 16.7 إنش -> 1 قدم و 4.7 إنش)

    return 0;
}

الأعضاء الاستاتيكية في الكلاس (Static Class Members)

بص يا هندسة، في الحالات الطبيعية لما بنعمل كلاس زي Student وبنخلق منه 3 كائنات، كل كائن بيكون عنده نسخته المستقلة تماماً من المتغيرات في الميموري. لو غيرنا اسم الكائن الأول، الباقي مش هيتأثر.
لكن تخيل لو إحنا عايزين نحتفظ بمتغير **"إجمالي عدد الطلاب المقبولين في الجامعة"**؟ مينفعش المتغير ده يكون تابع لكائن معين، لازم يكون **"مشترك"** بين كل الكائنات، ولو اتغير عند كائن يتغير عند الباقي تلقائياً! ده بقى دور الـ **Static Members**.

1. المتغيرات الاستاتيكية (Static Fields):

  • ذاكرة مشتركة (Shared Memory): المتغير الاستاتيكي بيتحجز في مكان واحد بس في الميموري لكل الكائنات المبنية على الكلاس.
  • دورة حياة البرنامج بأكمله (Entire Program Lifetime): بيفضل موجود في الذاكرة من أول ما البرنامج يشتغل لحد ما يقفل، حتى لو مفيش ولا كائن (Object) تم إنشاؤه من الكلاس!
  • قاعدة أكاديمية حرجة جداً: المتغير الاستاتيكي لازم يتعرف مرتين؛ المرة الأولى **جوة الكلاس** للإعلان عنه، والمرة الثانية **برة الكلاس بالكامل** لحجز مكان في الذاكرة وتهيئته (وإلا هيطلعلك **Linker Error**).

2. الدوال الاستاتيكية (Static Methods):

  • استدعاء على مستوى الكلاس: بنستدعيها باسم الكلاس نفسه مباشرة بالـ Scope Resolution Operator (`::`) بدون ما نضطر نعمل كائن! مثلاً: Student::getStudentCount();
  • القاعدة الذهبية للدوال الاستاتيكية: الدوال الاستاتيكية **متقدرش تكلم غير متغيرات استاتيكية تانية**! مينفعش تدخل جواها وتنادي متغير عادي (Instance Variable)، وده لأنها بتشتغل بدون وجود كائن فعلي فمينفعش تسألها عن داتا كائن معين!

الـ Static بالبلدي

الـ **Static Field** هو عامل زي "سبورة مشتركة" في مدرج الجامعة؛ أي دكتور يدخل المدرج يقدر يكتب عليها أو يعدل اللي مكتوب، والكل بيشوف نفس التعديل. بينما المتغير العادي بيمثل "كشكول الطالب الخاص"؛ كل طالب بيكتب في كشكوله ومحدش ليه دعوة ببيانات التاني!

تطبيق عملي: كلاس تتبع عدد الطلاب المقبولين في الجامعة

static_members.cpp
#include <iostream>
#include <string>
using namespace std;

class Student {
private:
    string name;
    // 1. الإعلان عن المتغير الاستاتيكي داخل الكلاس
    static int totalStudents; 

public:
    // المنشئ: بيزود العداد تلقائياً مع كل طالب جديد
    Student(string n) : name(n) {
        totalStudents++;
        cout << "[System] Student " << name << " enrolled successfully." << endl;
    }

    // المهدم: بينقص العداد تلقائياً لما الطالب يتخرج أو الكائن يتحذف
    ~Student() {
        totalStudents--;
        cout << "[System] Student " << name << " removed from memory." << endl;
    }

    // 2. دالة استاتيكية لقراءة العداد مباشرة باسم الكلاس
    static int getTotalStudents() {
        // ملحوظة: مينفعش أكتب هنا name = "test"; لأنها دالة static!
        return totalStudents;
    }
};

// 3. التعريف الإجباري وحجز الميموري للمتغير الاستاتيكي برة الكلاس
int Student::totalStudents = 0; 

int main() {
    cout << "--- Initial University Database Count: " << Student::getTotalStudents() << " ---" << endl;

    // عمل طلاب جدد
    Student* s1 = new Student("Ahmed");
    Student* s2 = new Student("Sara");
    Student* s3 = new Student("Omar");

    cout << "\nCurrent student count: " << Student::getTotalStudents() << endl; // هتطبع 3

    cout << "\n--- Deleting Sara (Sara Graduated) ---" << endl;
    delete s2; // تشغيل الـ Destructor لـ Sara وتخفيض العداد

    cout << "Current student count: " << Student::getTotalStudents() << endl; // هتطبع 2

    // تنظيف باقي الرام
    delete s1;
    delete s3;

    cout << "\nFinal count at program exit: " << Student::getTotalStudents() << endl; // هتطبع 0

    return 0;
}

شرح تفصيلي لطريقة التعريف الخارجي:

  • static int totalStudents; دي خطوة الإعلان فقط (Declaration)؛ بنعلم الـ compiler إن الكلاس يحتوي على هذا المتغير المشترك، ولكن لم يتم حجز أي بايت له في الرام بعد.
  • int Student::totalStudents = 0; دي أهم خطوة (Definition & Initialization)؛ بنحط نوع البيانات ثم اسم الكلاس متبوعاً بـ الـ Scope operator وعازل الميموري، وبنديله القيمة البدئية `0` خارج أي دالة.
  • Student::getTotalStudents() طريقة استدعاء الدالة الـ static؛ مش بنحتاج نعمل `s1.getTotalStudents()` (ولو إنه ينفع)، الأصح برمجياً نناديها باسم الكلاس مباشرة.

تحميل المعاملات الزائد (Operator Overloading)

بص يا هندسة، دي من أروع وأقوى مميزات الـ C++!
في الطبيعي لما بنكتب int z = x + y; المعالج بيفهم فوراً إن علامة الـ `+` هتجمع رقمين. ولو كتبنا string text = str1 + str2; الـ C++ بتفهم إن الـ `+` هنا هتعمل دمج للنصوص (Concatenation).
لكن لو عملنا كلاس للمسافات `Distance` وحاولنا نكتب Distance d3 = d1 + d2; المترجم هيطلع **Compile Error** ويقولك: **"أنا معرفش يعني إيه علامة جمع بين كائنين من نوع Distance!"**
وهنا بييجي دور الـ **Operator Overloading**؛ إحنا بنعيد تعريف المعامل الرياضي ونديله وظيفة إضافية ذكية يتعامل بيها مع كائنات الكلاس بتاعنا!

شروط وقواعد تحميل المعاملات:

  • الكلمة المفتاحية (Keyword): بنستخدم الكلمة الخاصة **`operator`** متبوعة بالعلامة اللي عايزين نحملها (مثلاً: `operator+` أو `operator-` أو `operator==`).
  • المعاملات الثنائية (Binary Operators): لما بنعمل overload لمعامل ثنائي زي الـ `+` جوة الكلاس، الدالة بتاخد **معامل واحد بس** في أقواسها؛ المعامل ده بيمثل الكائن اللي على اليمين، بينما الكائن اللي على اليسار هو اللي استدعى الدالة تلقائياً عن طريق مؤشر الـ this!
  • نوع الإرجاع (Return Type): يفضل دايماً إن الدالة ترجع كائن جديد تماماً بيحتوي على ناتج العملية البرمجية عشان نقدر نعمل Chain للعمليات (مثلاً: `d4 = d1 + d2 + d3;`).

الـ Operator Overloading بالبلدي

الموضوع عامل زي ما تكون شغال في شركة هندسية وبتعلم المترجم الغبي لغة مهندسين؛ بتقوله: "يا بني لما تلاقيني كاتب علامة زائد بين مسافتين، متقفلش البرنامج! خد القدم واجمعه على القدم، والإنش على الإنش، ولو الإنشات عدت الـ 12 حولها لقدام إضافية، ورجعلي المسافة النهائية الكلية!"

التطبيق الأكاديمي المثالي: كلاس المسافة مع تحميل الـ + والـ ==

operator_overloading.cpp
#include <iostream>
using namespace std;

class Distance {
private:
    int feet;
    float inches;

public:
    Distance() : feet(0), inches(0.0) {}
    Distance(int ft, float in) : feet(ft), inches(in) {}

    void show() const {
        cout << feet << " ft, " << inches << " in" << endl;
    }

    // 1. تحميل معامل الجمع الثنائي (+)
    // الكائن d2 ممرر كـ parameter (الجانب الأيمن للعلامة)
    Distance operator+ (const Distance &d2) {
        float totalInches = inches + d2.inches;
        int extraFeet = 0;

        if (totalInches >= 12.0) {
            extraFeet = (int)(totalInches / 12.0);
            totalInches = totalInches - (extraFeet * 12.0);
        }

        int totalFeet = feet + d2.feet + extraFeet;

        // إرجاع كائن جديد يحمل المجموع
        return Distance(totalFeet, totalInches);
    }

    // 2. تحميل معامل المقارنة الثنائي (==)
    bool operator== (const Distance &d2) {
        // بيقارن الطول الكلي بعد تحويله للإنشات
        float total1 = (feet * 12.0) + inches;
        float total2 = (d2.feet * 12.0) + d2.inches;
        return (total1 == total2);
    }
};

int main() {
    Distance d1(6, 8.5);  // 6 قدم و 8.5 إنش
    Distance d2(3, 5.5);  // 3 قدم و 5.5 إنش
    
    cout << "Distance 1: "; d1.show();
    cout << "Distance 2: "; d2.show();

    // استعمال علامة الجمع بشكل طبيعي جداً وسلس!
    // d1 استدعى الدالة، و d2 اتبعت كمعامل
    Distance d3 = d1 + d2; 
    
    cout << "\nResult of (d1 + d2): "; d3.show(); // هتطلع 10 قدم و 2 إنش (لأن 8.5 + 5.5 = 14 إنش -> 1 قدم و 2 إنش)

    // مقارنة المسافات باستعمال علامة التساوي
    Distance test1(2, 6.0);
    Distance test2(2, 6.0);

    cout << "\nComparing test1 and test2:" << endl;
    if (test1 == test2) {
        cout << "Yes! Distances are perfectly equal." << endl;
    } else {
        cout << "No! Distances are different." << endl;
    }

    return 0;
}

مقدمة في الوراثة (Introduction to Inheritance)

بص يا هندسة، الوراثة هي حجر الأساس وقوة الـ OOP الحقيقية!
تخيل لو إحنا شغالين في نظام مالي لشركة، وعايزين نعمل كلاس للـ **Employee** وكلاس للـ **Manager** وكلاس للـ **Developer**. التلاتة ليهم نفس الخصائص الأساسية زي الاسم، الرقم الوظيفي، والراتب الأساسي.
لو كتبنا المتغيرات دي جوة كل كلاس يدوياً، هنكرر كود كتير جداً ونهدر وقت وصيانة! الحل؟ بنعمل كلاس عام بنسميه **Base Class** (أو Parent Class) فيه الداتا المشتركة، وباقي الكلاسات بتورث منه وتتحول لـ **Derived Classes** (أو Child Classes)، ونزود في الكلاسات الابنة الخصائص الخاصة بيها بس!

مميزات الوراثة الذهبية:

  • إعادة استخدام الكود (Code Reusability): بنكتب الكود مرة واحدة في الأب، والكل بيورثه ويستخدمه مجاناً، وده بيوفر وقت وملايين الجنيهات في تطوير الأنظمة.
  • علاقة الـ "Is A" (The "is a" Relationship): الوراثة دايماً بتمثل علاقة منطقية حقيقية؛ مثلاً: الـ Manager هو في الأصل موظف (Manager **is an** Employee)، والسيارة هي مركبة (Car **is a** Vehicle).

محدد الوصول المحمي (protected:):

بص يا هندسة، إحنا عارفين الـ private (محدش بيشوفه غير الكلاس نفسه) والـ public (مفتوح للكل).
الـ C++ وفرت محدد تالت عبقري اسمه **`protected`**؛ وهو بيقول: **"المتغيرات المكتوبة هنا ممنوع أي كود خارجي في الـ main يلمسها، ولكن مسموح بالكامل لأي كلاس ابن وارث مني إنه يوصلها ويعدل فيها مباشرة!"**

جدول قواعد وأنماط الوراثة (Modes of Inheritance):

نوع العضو في الأب (Base Member) الوراثة العامة (Public mode) الوراثة المحمية (Protected mode) الوراثة الخاصة (Private mode)
public: بيبقى public في الابن. بيبقى protected في الابن. بيبقى private في الابن.
protected: بيبقى protected في الابن. بيبقى protected في الابن. بيبقى private في الابن.
private: مخفي تماماً: لا يمكن الوصول إليه مباشرة. مخفي تماماً: لا يمكن الوصول إليه مباشرة. مخفي تماماً: لا يمكن الوصول إليه مباشرة.

التطبيق الأكاديمي: وراثة كلاس الطالب من كلاس الشخص

simple_inheritance.cpp
#include <iostream>
#include <string>
using namespace std;

// 1. الكلاس الأساسي / الأب (Base Class)
class Person {
protected:
    // محمية: الكلاسات الابنة فقط تقدر تشوفها وتعدلها مباشرة
    string name;
    int age;

public:
    Person(string n, int a) : name(n), age(a) {
        cout << "[Base Constructor] Person created: " << name << endl;
    }

    void showPersonInfo() const {
        cout << "Name: " << name << " | Age: " << age << endl;
    }
};

// 2. الكلاس المشتق / الابن (Derived Class)
// يورث كلاس الشخص وراثة عامة (public)
class Student : public Person {
private:
    float gpa; // خاصية إضافية خاصة بالابن فقط

public:
    // المنشئ الخاص بالابن لازم ينادي منشئ الأب ويمررله المعاملات المطلوبة أولاً!
    Student(string n, int a, float g) : Person(n, a), gpa(g) {
        cout << "[Derived Constructor] Student details loaded." << endl;
    }

    void showStudentInfo() const {
        // نقدر نوصل لـ name و age مباشرة لأنهم protected في الأب!
        cout << "Student: " << name << " | Age: " << age << " | GPA: " << gpa << endl;
    }
};

int main() {
    cout << "--- Creating Person Object ---" << endl;
    Person p("Dr. Amr", 45);
    p.showPersonInfo();

    cout << "\n--- Creating Student Object ---" << endl;
    // هيتم استدعاء منشئ Person الأول، ثم منشئ Student!
    Student s("Zain", 20, 3.95);
    s.showStudentInfo();

    // اختبار أمان الـ protected:
    // s.name = "Test"; // هيطلع Compile Error لأن الـ main ليس ابناً للكلاس!

    return 0;
}

Professional Coding Challenges:

Challenge 1: Product Active Inventory Tracker
Design a class named **`Product`** containing product name and price, and a private static variable to track the total number of products currently in stock. When a product is created, increment the counter; when deleted, decrement it. Provide a static function to get the current inventory size.

بص يا هندسة، هنا بنزود العداد في الـ Constructor وبننقصه في الـ Destructor مع كتابة التعريف برة الكلاس:

#include <iostream>
#include <string>
using namespace std;

class Product {
private:
    string name;
    double price;
    static int activeProducts; // الإعلان عن عداد المنتجات النشطة

public:
    Product(string n, double p) : name(n), price(p) {
        activeProducts++;
    }

    ~Product() {
        activeProducts--;
    }

    // دالة استاتيكية لقراءة المخزون
    static int getActiveInventory() {
        return activeProducts;
    }
};

// تعريف وحجز مكان العداد الاستاتيكي
int Product::activeProducts = 0;

int main() {
    Product* p1 = new Product("Laptop", 1200.0);
    Product* p2 = new Product("Smartphone", 800.0);
    
    cout << "Active Inventory: " << Product::getActiveInventory() << endl; // هتطبع 2

Challenge 2: Complex Number Class with Operator Overloading
In mathematics, a complex number consists of a real part and an imaginary part ($a + bi$). Write a class named **`Complex`** that overloads the operator **`+`** to add two complex numbers together, and overloads the operator **`-`** to subtract them, and displays the result beautifully.

بص يا هندسة، في جمع الأعداد المركبة، بنجمع الجزء الحقيقي مع الحقيقي والتخيلي مع التخيلي:

#include <iostream>
using namespace std;

class Complex {
private:
    float real;
    float imag;

public:
    Complex() : real(0), imag(0) {}
    Complex(float r, float i) : real(r), imag(i) {}

    void display() const {
        cout << real << " + " << imag << "i" << endl;
    }

    // تحميل معامل الجمع (+)
    Complex operator+ (const Complex &c2) {
        float r = real + c2.real;
        float i = imag + c2.imag;
        return Complex(r, i);
    }

    // تحميل معامل الطرح (-)
    Complex operator- (const Complex &c2) {
        float r = real - c2.real;
        float i = imag - c2.imag;
        return Complex(r, i);
    }
};

int main() {
    Complex c1(4.5, 3.2); // 4.5 + 3.2i
    Complex c2(1.5, 1.8); // 1.5 + 1.8i

    cout << "Number 1: "; c1.display();
    cout << "Number 2: "; c2.display();

    Complex sum = c1 + c2;
    Complex diff = c1 - c2;

    cout << "\nResult of addition (c1 + c2): "; sum.display();  // هتطبع 6 + 5i
    cout << "Result of subtraction (c1 - c2): "; diff.display(); // هتطبع 3 + 1.4i

    return 0;
}
